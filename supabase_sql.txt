CREATE TABLE mensajes_foro (
    id BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    usuario TEXT NOT NULL,
    mensaje TEXT NOT NULL,
    fecha TIMESTAMP DEFAULT NOW()
);

-- Activar Row Level Security
ALTER TABLE mensajes_foro ENABLE ROW LEVEL SECURITY;

-- Permitir que cualquiera pueda leer mensajes
CREATE POLICY "Cualquiera puede leer mensajes"
ON mensajes_foro
FOR SELECT
USING (true);

-- Permitir insertar mensajes desde la web
CREATE POLICY "Usuarios pueden publicar mensajes"
ON mensajes_foro
FOR INSERT
WITH CHECK (true);
